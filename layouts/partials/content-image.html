{{/* Render a content image. PNG and JPEG images that resolve to a resource are served as with a WebP derivative, keeping the original as the fallback . Anything else is rendered as a plain . @param {dict} . - A dictionary containing: - "page" {page}: The page the image appears on. - "src" {string}: The image destination as written in Markdown. - "alt" {string}: The alternative text. - "class" {string}: The CSS classes for the . Set params.academy.webp.enable to false to skip WebP encoding. A missing param means enabled. (Not params.images: Hugo reads that as the list of default social card images.) WebP is encoded at q85 because Hugo's WebP encoder is lossy only: at q100 the output is larger than the source PNG, and q85 keeps screenshot text sharp. */}} {{- $page := .page -}} {{- $src := .src -}} {{- $alt := .alt -}} {{- $class := .class -}} {{- $enabled := true -}} {{- with site.Params.academy -}} {{- with .webp -}} {{- if or (eq .enable false) (eq .enable "false") -}} {{- $enabled = false -}} {{- end -}} {{- end -}} {{- end -}} {{- $img := "" -}} {{- $webp := "" -}} {{- if and $enabled hugo.IsExtended -}} {{- $img = partial "resolve-content-image.html" (dict "page" $page "src" $src) -}} {{- with $img -}} {{- with try (.Process "webp q85") -}} {{- if .Err -}} {{- warnf "content-image: could not encode %q to WebP on %q, serving the original: %s" $src $page.RelPermalink .Err -}} {{- else -}} {{- $webp = .Value -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $webp -}} {{- else -}} {{- end -}}
